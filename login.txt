    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-app.js";
        import { getAuth, signInWithEmailAndPassword, createUserWithEmailAndPassword, setPersistence, browserLocalPersistence } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-auth.js";

        const firebaseConfig = {
            apiKey: "AIzaSyC1G1VWM_0sIaLQhGtrvmG2xPfJpFHzU70",
            authDomain: "autolux-8aa0d.firebaseapp.com",
            projectId: "autolux-8aa0d",
            storageBucket: "autolux-8aa0d.firebasestorage.app",
            messagingSenderId: "902279691304",
            appId: "1:902279691304:web:2afd537763ce1117a3891d",
            measurementId: "G-ZMM7HY0798"
        };

        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);

        let modoRegistro = false;

        // Cambiar entre modo login y registro
        window.toggleModo = () => {
            modoRegistro = !modoRegistro;
            const btn = document.getElementById('btn-auth');
            const link = document.getElementById('toggle-auth');
            const title = document.getElementById('auth-title');

            if (modoRegistro) {
                title.innerText = "CREAR CUENTA";
                btn.innerText = "REGISTRARSE";
                link.innerText = "¿Ya tienes cuenta? Inicia sesión";
            } else {
                title.innerText = "AUTOLUX";
                btn.innerText = "INICIAR SESIÓN";
                link.innerText = "¿No tienes cuenta? Regístrate aquí";
            }
        };

        window.ejecutarAuth = async () => {
            const email = document.getElementById('email').value;
            const pass = document.getElementById('pass').value;
            const msg = document.getElementById('error-msg');
            const btn = document.getElementById('btn-auth');

            if (!email || !pass) {
                msg.innerText = "Completa los campos.";
                msg.classList.remove('hidden');
                return;
            }

            btn.disabled = true;
            const originalText = btn.innerText;
            btn.innerText = "PROCESANDO...";

            try {
                await setPersistence(auth, browserLocalPersistence);
                
                if (modoRegistro) {
                    // REGISTRO
                    await createUserWithEmailAndPassword(auth, email, pass);
                } else {
                    // LOGIN
                    await signInWithEmailAndPassword(auth, email, pass);
                }
                
                window.location.replace("index.html");
                
            } catch (error) {
                console.error("Error:", error.code);
                msg.classList.remove('hidden');
                btn.disabled = false;
                btn.innerText = originalText;
                
                // Traducción de errores comunes
                switch (error.code) {
                    case 'auth/email-already-in-use':
                        msg.innerText = "Este correo ya está registrado.";
                        break;
                    case 'auth/weak-password':
                        msg.innerText = "La contraseña debe tener al menos 6 caracteres.";
                        break;
                    case 'auth/invalid-credential':
                        msg.innerText = "Datos incorrectos.";
                        break;
                    default:
                        msg.innerText = "Error de conexión o datos inválidos.";
                }
            }
        };
    </script>
